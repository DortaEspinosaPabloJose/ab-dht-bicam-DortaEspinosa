99 Boolean found = false;
100 List<String> key = Strategy.getStrategy().getListKey();
101 int count = 0;
102 if(RandomSearch.listStateReference.size() == 0){
103 return this.setListParticle(new ArrayList<Particle>());

105 while((found.equals(false)) && (Strategy.getStrategy().mapGenerators.size() > count)){
106 //recorrer la lista de generadores, hasta que encuentre el PSO
107 if(key.get(count).equals(GeneratorType.ParticleSwarmOptimization.toString())){
108 //creo el generador PSO, y si su lista de particulas esta vacia entonces es la primera vez que lo estoy creando, y cada estado lo convierto en particulas
109 GeneratorType keyGenerator = GeneratorType.valueOf(String.valueOf(key.get(count)));
110 ParticleSwarmOptimization generator = (ParticleSwarmOptimization) Strategy.getStrategy().mapGenerators.get(keyGenerator);
111 if(generator.getListParticle().isEmpty()){
112 //convertir los estados en particulas
113 for (int j = 0; j < RandomSearch.listStateReference.size(); j++) {
114 //si el estado es creado con el generator RandomSearch entonces la convierto en particula
115 if(getListParticle().size() != countRef){
116 ArrayList<Object> velocity = new ArrayList<Object>();
117 State stateAct = (State) RandomSearch.listStateReference.get(j).getCopy();
118 stateAct.setCode(new ArrayList<Object>(RandomSearch.listStateReference.get(j).getCode()));
119 stateAct.setEvaluation(RandomSearch.listStateReference.get(j).getEvaluation());
120
121 State statePBest = (State) RandomSearch.listStateReference.get(j).getCopy();
122 statePBest.setCode(new ArrayList<Object>(RandomSearch.listStateReference.get(j).getCode()));
123 statePBest.setEvaluation(RandomSearch.listStateReference.get(j).getEvaluation());
124
125 Particle particle = new Particle(stateAct, statePBest, velocity);
126 getListParticle().add(particle);
127 }
128 }
129 }
130 else{
131 setListParticle(generator.getListStateReference());
132 }
133 found = true;
134 }
135 count++;
136 }
